import {readFileSync,writeFileSync,mkdirSync} from 'node:fs';
import {makeLargeHouseFixture} from '../../demo/fixtures/large-house.mjs';
import {migrateGraphSpace} from '../../test-build/plan-graph.js';
const dir=new URL('./ha/.storage/',import.meta.url);
const read=name=>JSON.parse(readFileSync(new URL(name,dir),'utf8'));
const write=(name,value)=>writeFileSync(new URL(name,dir),JSON.stringify(value,null,2)+'\n');
const fixture=makeLargeHouseFixture({floors:1,roomsPerFloor:14,devices:26,openings:19,partitions:0,columns:0,decor:12,includeStairs:false});
const devices=read('core.device_registry').data.devices.filter(d=>!d.disabled_by);
const selected=devices.slice(0,26);
const areas=read('core.area_registry').data.areas;
fixture.config.spaces[0].rooms.forEach((r,i)=>r.area=areas[i%areas.length].id);
fixture.config.spaces=fixture.config.spaces.map(space=>migrateGraphSpace(space).space);
fixture.config.model_version=11;
fixture.config.settings={...fixture.config.settings,filter_seeded:true,volumetric_view:false,group_lights:false};
fixture.config.markers=devices.map((d,i)=>({id:d.id,binding:`device:${d.id}`,hidden:i>=26}));
const positions=Object.values(fixture.layout);
const layout=Object.fromEntries(selected.map((d,i)=>[d.id,positions[i]]));
const config=read('houseplan.config');config.data.config=fixture.config;config.data.rev=1;write('houseplan.config',config);
const stored=read('houseplan.layout');stored.data.layout=layout;stored.data.rev=1;write('houseplan.layout',stored);
const dashboard=read('lovelace.house_plan');dashboard.data.config={views:[{title:'Profile',path:'profile',type:'masonry',cards:[{type:'custom:houseplan-card'}]}]};write('lovelace.house_plan',dashboard);
const scene={...fixture.counts,model:11,sourceRoomEdges:56,physicalSegments:fixture.config.spaces[0].wall_segments?.length,source:'large-house.mjs with actual HA seed bindings',space:fixture.config.spaces[0].id};
mkdirSync(new URL('./',import.meta.url),{recursive:true});writeFileSync(new URL('./scene.json',import.meta.url),JSON.stringify(scene,null,2));console.log(scene);
